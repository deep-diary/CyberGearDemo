# 1 功能切换

电机控制包含七种控制模式，通过 UserAppID 变量显示当前运行模式，通过RequestedUserAppID赋值不同切换不同控制模式。，例如RequestedUserAppID默认是0x00，为USER_APP_NONE模式，RequestedUserAppID = 0x01切换位置控制模式。

```c
typedef enum {
  USER_APP_NONE, //上电后默认模式，包含速度控制和电流控制
  USER_APP_NORMAL_POS_CTRL, //位置控制模式，包含扫频和位置控制
  USER_APP_ENCODER_ALIGNMENT, //磁编校准模式
  USER_APP_CURRENTLOOP_BW_TEST, //电流环环扫频模式
  USER_APP_SPEEDLOOP_BW_TEST,   //速度环扫频模式
  USER_APP_JOG,
  USER_APP_MIT_CONTROL, //运动控制模式
  USER_APP_COUNT,
} USER_APP_ID;
```

# 2 USER_APP_NON模式

控制字 BasicApp_Handle_t BasicApp

```c
typedef struct {
  UserApplication_Handle_t _Super;
  
  int16_t Iref;//电流指令
  int16_t SpeedRef;//速度指令
  int16_t PrevSpeedRef;
  int16_t Durationms;//指令斜坡时间，默认1S

  union UFlags
  {
    uint16_t all;
    struct
    {
      uint16_t MotorOn : 1;
      uint16_t FaultReset : 1;
    } bits;
  } Flags;
  

} BasicApp_Handle_t;
```

切换USER_APP_NON后默认为速度模式，通过给BasicApp._Super.pMCI.lastCommand赋值切换控制模式，0x03为速度模式，0x04为转矩模式，然后BasicApp.flags.all赋值0x01启动

# 3 USER_APP_NORMAL_POS_CTRL模式

控制字 PositionCtrlApp_Handle_t PositionCtrolApp
```c
typedef struct {
  UserApplication_Handle_t _Super;
  SineGenerator_Handle_t*  pSin;
  PositionProfileGenerator_Handle_t* pPosGen;

  int32_t PosRef;
  int32_t PrevPosRef;

  uint16_t RefSinEndFreq01Hz;
  uint16_t SweepDuration_ms;


  union 
  {
    uint16_t all;
    struct
    {
      uint16_t MotorOn : 1;

# 4 USER_APP_ENCODER_ALIGNMENT
偏心校准流程和之前的磁编校准相同，先切换模式再使能
```c
控制字 EncoderAlignmentApp_Handle_t EncoderAlignmentApp

typedef struct {
  UserApplication_Handle_t _Super;
  ENCODER_Handle_t*        pEncoder;
  PositionProfileGenerator_Handle_t* pPosGen;

  uint32_t TimeStamp;
  uint32_t AlignmentDuration;
  uint32_t SpinningDuration;
  uint16_t AlignedMecAngle;
  int32_t IdrefStep;
  int32_t PosAccumulator;
  int16_t Idref;
  int16_t error;

  union {
    uint16_t all;
    struct {
      uint16_t Start         : 1;//校准开始
      uint16_t EnableSave2EE : 1;//校准结果存储
      uint16_t Rsvd          : 14;
    } bits;

  } flags;

  EAA_STATE_t state;
} EncoderAlignmentApp_Handle_t;
使能时，flags.all 赋值3，即可开始校准并在校准完成后自动存储校准结果

检查校准结果：EncoderAlignmentApp.pEncoder.hAngleError数组是否为0，是否和flash 地址0x0801F800中的数据保持一致
相关变量
    ENCODER_M1.CalibrationCompletedFlag   此次校准启动后校准是否完成
    ENCODER_M1.iSCalibrationCompletedFlag 该驱动器是否曾完成过校准

# 6 USER_APP_MIT_CONTROL
RequestedUserAppID 切换06，可运行运动控制模式

控制字
```c
MITControlApp_Handle_t MITControlApp = {
    ._Super =
        {
            .pMCI                     = &Mci[M1],
            .pFctInit                 = NULL,
            .pFctReset                = MITControlApp_OnReset,
            .pFctOnStart              = NULL,
            .pFctOnExit               = MITControlApp_OnExit,
            .pFctPreLowFreqUpdate     = MITControlApp_OnLowFrequencyUpdate,
            .pFctPostLowFreqUpdate    = NULL,
            .pFctPreMediumFreqUpdate  = NULL,
            .pFctPostMediumFreqUpdate = NULL,
            .pFctPreHighFreqUpdate    = NULL,
            .pFctPostHighFreqUpdate   = NULL,
            .pFctBackgroundUpdate     = MITControlApp_OnBackground,
            .Activated                = false,
            .OneShootTask             = false,
            .OneShootTaskFinished     = false,
        },
    .PosRef           = 0,//代表减速前绝对位置，65535代表电机减速前一圈
    .VelRef           = 0,//电机运行速度，单位rad/s
    .Kp               = 100,  //4000代表1

# 7 设置零位和回零功能

```c
/* 回零和零点设置功能结构体变量定义 */
HomingControl_t s_HomingControl = {
    .homingModeFlag = 0,     // 回零模式标志位
    .homingStartFlag = 0,    // 回零开始标志位
    .homingCounter = 0,      // 回零计数器
    .posRefSetSuccessFlag = 0, // PosRef设置成功标志位
    .posRefSetExecuted = 0,  // PosRef设置已执行标志位，防止重复执行
    .setZeroFlag = 0,        // 设置零点标志位
};
```
在任意模式和位置都可使用setZeroFlag去设置当前位置为0位
homingModeFlag切换为回零模式，homingStartFlag开始回零
